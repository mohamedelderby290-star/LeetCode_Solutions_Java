public class Main {
    public static void main (String[] args) {

        int [] flowerbed = {1,0,0,0,0,1,0,};
        int n = 1;
        boolean canPlace = canPlaceFlowerbed(flowerbed, n);
        System.out.println ("Can place   "  + n + " flowerbeds "  +  canPlace);
    }

    public static boolean canPlaceFlowerbed(int [] flowerbed, int n){
        for(int i = 0 ; i < flowerbed.length; i++){
            if(flowerbed[i] == 0
            && (i ==0 || flowerbed[i-1] == 0 )
            && (i == flowerbed.length - 1 || flowerbed[ i+ 1] ==0)){

                flowerbed[i]= 1;
                n--;
            }
        }
        return n <= 0;
    }
}
