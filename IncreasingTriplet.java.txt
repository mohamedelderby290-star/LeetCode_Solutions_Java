public class Main {
    public static void main (String[] args) {
        Solution sol = new Solution();
        int [] nums = {1,2,3,4,5};
        boolean result1 = sol.IncreasingTriplet(nums);
        System.out.println ("Input Increasing : [1,2,3,4,5] => Output :" + result1);

        int [] nums2 = {5,4,3,2,1};
        boolean result2 = sol.IncreasingTriplet(nums2);
        System.out.println ("Input Increasing : [5,4,3,2,1] => Output :" + result2);

    }
}
public class Solution {
    public boolean IncreasingTriplet(int [] nums){
        int first = Integer.MAX_VALUE;
        int second = Integer.MAX_VALUE;
        for (int num : nums){
            if(num <= first){
                first = num ;
            } else if (num <= second) {
                second = num ;

            }
            else {
                return true;
            }
        }
        return false ;
    }
}
